def test_add_merchant(api):
    payload = {
        "id": 0,
        "merchantName": "Pytest Merchant",
        "merchantId": "pytest-001",
        "idDescription": 1
    }

    response = api.post(
        "/merchant",
        json=payload
    )

    assert response.status_code == 200

    body = response.json()

    assert "merchants" in body

    assert any(
        merchant["merchantId"] == "pytest-001"
        for merchant in body["merchants"]
    )


def test_delete_merchant(api):
    payload = {
        "id": 0,
        "merchantName": "Delete Test",
        "merchantId": "delete-test",
        "idDescription": 1
    }

    response = api.post(
        "/merchant",
        json=payload
    )

    body = response.json()

    merchant = next(
        x for x in body["merchants"]
        if x["merchantId"] == "delete-test"
    )

    response = api.delete(
        "/merchant",
        json=merchant
    )

    assert response.status_code == 200

    body = response.json()

    assert not any(
        x["merchantId"] == "delete-test"
        for x in body["merchants"]
    )

def change_password(self, username, password, new_password):
    response = self.session.post(
        f"{self.base_url}/changepassword",
        headers={
            "username": username,
            "password": password,
            "newpassword": new_password,
            "Content-Length": "0"
        },
    )

    response.raise_for_status()

    self.token = response.json()["token"]

    return response


import os

def test_change_password(api):
    username = os.getenv("USERNAME")
    old_password = os.getenv("PASSWORD")
    new_password = "432109876573aZ"

    try:
        response = api.change_password(
            username=username,
            password=old_password,
            new_password=new_password,
        )

        assert response.status_code == 200
        assert "token" in response.json()

    finally:
        # Всегда пытаемся вернуть старый пароль
        api.change_password(
            username=username,
            password=new_password,
            new_password=old_password,
        )
