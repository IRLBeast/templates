import base64

def change_password(self, username, password, new_password):
    headers = {
        "username": base64.b64encode(username.encode()).decode(),
        "password": base64.b64encode(password.encode()).decode(),
        "newpassword": base64.b64encode(new_password.encode()).decode(),
    }

    response = self.session.post(
        f"{self.base_url}/changepassword",
        headers=headers,
    )

    response.raise_for_status()

    self.token = response.json()["token"]

    return response


import os

def test_change_password(api):
    username = os.getenv("USERNAME")
    old_password = os.getenv("PASSWORD")
    new_password = os.getenv("NEW_PASSWORD")

    try:
        response = api.change_password(
            username=username,
            password=old_password,
            new_password=new_password,
        )

        assert response.status_code == 200
        assert "token" in response.json()

    finally:
        # Возвращаем старый пароль
        api.change_password(
            username=username,
            password=new_password,
            new_password=old_password,
        )
